Completely replace the contents of `app/(auth)/login/page.tsx` with the following file. This preserves all behavior (state, submit handler, error handling, redirect) but switches the UI to shadcn primitives so it automatically inherits the new theme.

Write this exact content:

```tsx
'use client'
import { useState } from 'react'
import { useRouter } from 'next/navigation'
import {
  Card,
  CardContent,
  CardDescription,
  CardFooter,
  CardHeader,
  CardTitle,
} from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'

export default function LoginPage() {
  const router = useRouter()
  const [password, setPassword] = useState('')
  const [error, setError] = useState('')
  const [loading, setLoading] = useState(false)

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    setLoading(true)
    setError('')
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ password }),
    })
    setLoading(false)
    if (res.ok) {
      router.push('/')
    } else {
      setError('Şifre hatalı. Lütfen tekrar deneyin.')
    }
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-background text-foreground p-4">
      <Card className="w-full max-w-[400px]">
        <CardHeader>
          <CardTitle>Sigorta Takip</CardTitle>
          <CardDescription>
            Avukat paneline erişmek için şifrenizi girin.
          </CardDescription>
        </CardHeader>
        <form onSubmit={handleSubmit}>
          <CardContent className="space-y-4">
            <div className="space-y-2">
              <Label htmlFor="password">Şifre</Label>
              <Input
                id="password"
                type="password"
                placeholder="••••••••"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                required
                autoFocus
              />
            </div>
            {error && (
              <p className="text-sm text-destructive" role="alert">
                {error}
              </p>
            )}
          </CardContent>
          <CardFooter>
            <Button type="submit" disabled={loading} className="w-full">
              {loading ? 'Giriş yapılıyor...' : 'Giriş Yap'}
            </Button>
          </CardFooter>
        </form>
      </Card>
    </div>
  )
}
```

Rules:
- Preserve the same POST target `/api/auth/login`, same payload `{ password }`, same redirect target `/`, same error message `'Şifre hatalı. Lütfen tekrar deneyin.'` — this keeps the existing auth flow and Phase 1 middleware behavior working.
- Use `bg-background text-foreground` on the outer div so the page respects the warm white palette from plan 08-01.
- Use the `text-destructive` Tailwind utility for the error text — this picks up the `--destructive` token automatically.
- Do NOT add any `style={{ ... }}` inline props. Do NOT use `backgroundColor`, `borderColor`, or `outlineColor` inline overrides.
- Do NOT import anything else (no icons, no toast, no form schemas).
- The file must be a default export named `LoginPage`.
