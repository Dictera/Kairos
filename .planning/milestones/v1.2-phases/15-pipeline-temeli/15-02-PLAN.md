    The component must:
    - Be a `'use client'` component that uses `useTRPC()` hook to call `trpc.pipeline.healthCheck.useQuery()` with `refetchInterval: 5 * 60 * 1000` (5-minute polling matching cache TTL)
    - Use React Query's `staleTime` and `gcTime` to avoid excessive refetching
    - Only render when data is loaded AND either `python.accessible === false` OR `libreoffice.accessible === false`
    - Use yellow/amber warning styling (`bg-amber-50 dark:bg-amber-950/50`, `border-amber-200 dark:border-amber-800`, `text-amber-800 dark:text-amber-200`) consistent with existing project patterns

    - Include a dismiss button (X icon) that hides the banner for the current session (use `useState` — banner reappears on page refresh)
    - The banner should appear at the very top of the main content area, above all page content
    - When loading (query pending), render null — don't show a loading state or skeleton for the banner
    - Import from `lucide-react`: `AlertTriangle`, `X` icons
    - The component should never crash — wrap the query in error handling, if query fails, don't render (fail silently)

    ```tsx
    // Key structure:
    'use client'
    import { useState } from 'react'
    import { useTRPC } from '@/lib/trpc/context'
    import { useQuery } from '@tanstack/react-query'
    import { AlertTriangle, X } from 'lucide-react'

    export function HealthBanner() {
      const trpc = useTRPC()
      const [dismissed, setDismissed] = useState(false)
      const { data, isLoading } = useQuery(
        trpc.pipeline.healthCheck.queryOptions({ refetchInterval: 5 * 60 * 1000 })
      )

      if (dismissed || isLoading || !data) return null

      const missingPython = !data.python.accessible
      const missingLibre = !data.libreoffice.accessible

      if (!missingPython && !missingLibre) return null

      // Build message based on what's missing
      // Render amber warning banner with dismiss button
    }
    ```
